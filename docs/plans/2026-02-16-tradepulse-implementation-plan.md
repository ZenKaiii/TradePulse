# src/tradepulse/ingest/rss.py
import hashlib
from tradepulse.models import CanonicalArticle

def parse_rss_entries(source_name: str, entries: list[dict]) -> list[CanonicalArticle]:
    out = []
    for e in entries:
        raw = f"{e.get('title','')}|{e.get('link','')}"
        aid = hashlib.sha1(raw.encode("utf-8")).hexdigest()
        out.append(
            CanonicalArticle(
                id=aid,
                title=e.get("title", "").strip(),
                url=e.get("link", "").strip(),
                source_name=source_name,
                published_at=e.get("published", ""),
            )
        )
    return out
```

@dataclass
class EventCluster:
    cluster_id: str
    articles: list[CanonicalArticle]
    coverage_count: int

def cluster_articles(items: list[CanonicalArticle]) -> list[EventCluster]:
    buckets: dict[str, list[CanonicalArticle]] = {}
    for it in items:
        buckets.setdefault(it.url, []).append(it)
    return [EventCluster(cluster_id=k, articles=v, coverage_count=len(v)) for k, v in buckets.items()]

from dataclasses import dataclass

@dataclass
class RuleScoreResult:
    rule_score: float
    direction: str
    affected_tickers: list[str]

def score_cluster(text: str, coverage_count: int) -> RuleScoreResult:
    t = text.lower()
    tickers = []
    if "nvidia" in t or "nvda" in t:
        tickers.append("NVDA")
    if any(x in t for x in ["raises guidance", "beats", "surge"]):
        direction = "bullish"
    elif any(x in t for x in ["cuts guidance", "misses", "probe", "ban", "lawsuit"]):
        direction = "bearish"
    else:
        direction = "neutral"
    score = min(10.0, 2.0 + coverage_count * 1.5 + (1.5 if direction != "neutral" else 0.0))
    return RuleScoreResult(rule_score=score, direction=direction, affected_tickers=tickers)
```

def _map_direction(direction: str) -> str:
    return {"bullish": "利好", "bearish": "利空", "neutral": "中性"}.get(direction, "中性")

def compose_digest(top_events: list[dict], overlays: dict) -> str:
    lines = ["# TradePulse 每小时快报", "## A. 关键事件 Top10"]
    for i, e in enumerate(top_events, 1):
        stocks = ", ".join([f"{x['symbol']} ({x['name']})" for x in e.get("affected_tickers", [])]) or "未识别"
        lines.append(f"{i}. {e['title']}")
        lines.append(f"   - 市场方向: {_map_direction(e['direction'])}")
        lines.append(f"   - 相关标的: {stocks}")
        lines.append(f"   - 影响说明: {e.get('impact_reason_zh', '暂无')}")
        for s in e.get("sources", []):
            lines.append(f"   - 来源: {s['name']} {s['url']}")
    return "\n".join(lines)
```

import sqlite3
from pathlib import Path

class PushLedger:
    def __init__(self, path: Path):
        self.conn = sqlite3.connect(path)
        self.conn.execute("create table if not exists pushed(cluster_id text primary key, run_id text, pushed_at text)")
        self.conn.commit()

    def should_push(self, cluster_id: str) -> bool:
        row = self.conn.execute("select 1 from pushed where cluster_id = ?", (cluster_id,)).fetchone()
        return row is None

    def mark_pushed(self, cluster_id: str, run_id: str) -> None:
        self.conn.execute("insert or ignore into pushed(cluster_id, run_id, pushed_at) values(?, ?, datetime('now'))", (cluster_id, run_id))
        self.conn.commit()
```
